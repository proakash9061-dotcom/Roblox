local Util = {}

function Util.isAlive(p)
    local c = p.Character
    if not c then return false end
    local h = c:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

function Util.root(p)
    local c = p.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

function Util.part(p, mode)
    local c = p.Character
    if not c then return nil end
    if mode == "Random" or mode == "Nearest" then
        local parts = { "Head", "UpperTorso", "LowerTorso", "LeftUpperArm", "RightUpperArm" }
        local best, bestD = nil, math.huge
        for _, n in ipairs(parts) do
            local part = c:FindFirstChild(n)
            if part then
                local d = (part.Position - Camera.CFrame.Position).Magnitude
                if d < bestD then best, bestD = part, d end
            end
        end
        return best
    elseif mode == "Head" then
        return c:FindFirstChild("Head")
    elseif mode == "Torso" then
        return c:FindFirstChild("UpperTorso")
            or c:FindFirstChild("Torso")
            or c:FindFirstChild("HumanoidRootPart")
    end
    return c:FindFirstChild("Head")
end

function Util.team(p)
    return p.Team ~= nil and p.Team == LP.Team
end

function Util.visible(part)
    if not part then return false end
    local origin = Camera.CFrame.Position
    local dir = part.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LP.Character, Camera }
    params.IgnoreWater = true
    local hit = Workspace:Raycast(origin, dir, params)
    return hit == nil or hit.Instance:IsDescendantOf(part.Parent)
end

-- TARGET MANAGER
-- ============================================================
local Target = {}
Target.locked = nil

function Target.acquire(fov, teamCheck, wallCheck, mode, scope)
    scope = scope or "Nearest"
    local best, bestScore = nil, math.huge
    local cx, cy = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LP or not Util.isAlive(p) then continue end
        if teamCheck and Util.team(p) then continue end

        local part = Util.part(p, mode)
        if not part then continue end

        if scope == "Nearest" then
            local sp, on = Camera:WorldToViewportPoint(part.Position)
            if not on then continue end
            local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
            if d > fov then continue end
            if wallCheck and not Util.visible(part) then continue end
            if d < bestScore then best, bestScore = p, d end
        else
            -- Server scope: any distance, still FOV-filtered if fov > 0
            local root = Util.root(p)
            if not root then continue end
            local d = (root.Position - Camera.CFrame.Position).Magnitude
            if fov > 0 then
                local sp, on = Camera:WorldToViewportPoint(part.Position)
                if not on then continue end
                local sd = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                if sd > fov then continue end
            end
            if wallCheck and not Util.visible(part) then continue end
            if d < bestScore then best, bestScore = p, d end
        end
    end
    return best
end

function Target.predict(part, root, amount)
    local pos = part.Position
    if root and amount > 0 then
        pos = pos + root.AssemblyLinearVelocity * amount
    end
    return pos
end

-- SILENT AIM (guarded + modern Raycast + legacy Ray)
-- ============================================================
local hookBusy = false

local ok, mt = pcall(getrawmetatable, game)
if ok and mt then
    local oldNamecall = mt.__namecall
    setreadonly(mt, false)

    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()

        if hookBusy
           or not Config.SilentAim.Enabled
           or checkcaller()
           or typeof(method) ~= "string" then
            return oldNamecall(self, ...)
        end

        local isLegacy = method:sub(1, 13) == "FindPartOnRay"
        local isModern = (method == "Raycast")

        if not isLegacy and not isModern then
            return oldNamecall(self, ...)
        end

        -- Modern Raycast must be called on Workspace
        if isModern and not (self == Workspace or (typeof(self) == "Instance" and self:IsA("Workspace"))) then
            return oldNamecall(self, ...)
        end

        if Config.SilentAim.HitChance < 100
           and math.random(1, 100) > Config.SilentAim.HitChance then
            return oldNamecall(self, ...)
        end

        hookBusy = true
        local pOk, ret = pcall(function()
            local t = Target.acquire(
                Config.SilentAim.FOV,
                Config.SilentAim.TeamCheck,
                Config.SilentAim.WallCheck,
                Config.SilentAim.Mode,
                "Nearest"
            )
            if not t then return oldNamecall(self, ...) end

            local part = Util.part(t, Config.SilentAim.Mode)
            local root = Util.root(t)
            if not part then return oldNamecall(self, ...) end

            local aimPos = Target.predict(part, root, Config.SilentAim.Prediction)
            local args = { ... }

            if isLegacy then
                local oldRay = args[1]
                if typeof(oldRay) == "Ray" then
                    args[1] = Ray.new(oldRay.Origin, aimPos - oldRay.Origin)
                elseif typeof(oldRay) == "table" and oldRay.Origin then
                    oldRay.Direction = aimPos - oldRay.Origin
                end
            elseif isModern then
                local origin, dir = args[1], args[2]
                if typeof(origin) == "Vector3" then
                    local nd = aimPos - origin
                    if typeof(dir) == "Vector3" and dir.Magnitude > 0 then
                        nd = nd.Unit * dir.Magnitude
                    end
                    args[2] = nd
                end
            end

            return oldNamecall(self, unpack(args))
        end)

        hookBusy = false
        if pOk then return ret end
        return oldNamecall(self, ...)
    end)
    setreadonly(mt, true)
end

local currentAimTarget = nil

RunService.RenderStepped:Connect(function()
    if not Config.Aimbot.Enabled then currentAimTarget = nil; return end

    -- Sticky retention
    if Config.Aimbot.StickyAim and currentAimTarget and Util.isAlive(currentAimTarget) then
        local part = Util.part(currentAimTarget, Config.Aimbot.Mode)
        if part and (not Config.Aimbot.WallCheck or Util.visible(part)) then
            -- keep target
        else
            currentAimTarget = nil
        end
    else
        currentAimTarget = nil
    end

    if not aimHeld then return end

    if not currentAimTarget then
        currentAimTarget = Target.acquire(
            Config.Aimbot.FOV, Config.Aimbot.TeamCheck,
            Config.Aimbot.WallCheck, Config.Aimbot.Mode, "Nearest"
        )
    end
    if not currentAimTarget then return end

    local part = Util.part(currentAimTarget, Config.Aimbot.Mode)
    local root = Util.root(currentAimTarget)
    if not part then return end

    local aimPos = Target.predict(part, root, Config.Aimbot.Prediction)
    local desired = CFrame.new(Camera.CFrame.Position, aimPos)
    Camera.CFrame = Camera.CFrame:Lerp(desired, 1 - Config.Aimbot.Smoothness)
end)

local Farm = {
    target = nil,
    state = "idle",
    timer = 0,
    shots = 0,
    orbitAngle = 0,
}

local function fireWeapon()
    -- Priority 1: VirtualInputManager (executor-provided)
    if VirtualInputManager then
        pcall(function()
            VirtualInputManager:SendMouseButtonEvent(
                Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2,
                0, true, game, 1
            )
            VirtualInputManager:SendMouseButtonEvent(
                Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2,
                0, false, game, 1
            )
        end)
        return true
    end
    -- Priority 2: mousemoverel + click via LP mouse
    if mousemoverel then
        pcall(function()
            mousemoverel(0, 0)
            Mouse:Click()
        end)
        return true
    end
    -- Priority 3: tool Activate
    local char = LP.Character
    if char then
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then
            pcall(function() tool:Activate() end)
            return true
        end
    end
    return false
end

local function flyStep(myRoot, dest, speed, dt)
    local delta = dest - myRoot.Position
    local dist = delta.Magnitude
    if dist < 0.15 then return true end
    local step = math.min(dist, speed * dt)
    myRoot.CFrame = CFrame.new(myRoot.Position + delta.Unit * step, dest)
    return false
end

RunService.Heartbeat:Connect(function(dt)
    if not Config.Farm.Enabled then
        Farm.target = nil
        Farm.state = "idle"
        return
    end

    local myChar = LP.Character
    local myRoot = Util.root(LP)
    if not myChar or not myRoot then return end
    local myHum = myChar:FindFirstChildOfClass("Humanoid")
    if myHum then myHum.PlatformStand = true end

    -- Acquire or re-acquire
    if not Farm.target or not Util.isAlive(Farm.target) then
        Farm.target = Target.acquire(
            0, Config.SilentAim.TeamCheck, false, Config.Farm.Mode == "Snipe" and "Head" or "Torso",
            Config.Farm.TargetScope
        )
        Farm.state = "approach"
        Farm.shots = 0
        Farm.timer = 0
    end
    if not Farm.target then return end

    local tRoot = Util.root(Farm.target)
    local tHead = Util.part(Farm.target, "Head")
    if not tRoot or not tHead then Farm.target = nil; return end

    -- Range guard
    local myDist = (tRoot.Position - myRoot.Position).Magnitude
    if myDist > Config.Farm.MaxRange then Farm.target = nil; return end

    -- Camera
    if Config.Farm.CameraFollow then
        local look = CFrame.new(Camera.CFrame.Position, tHead.Position)
        Camera.CFrame = Camera.CFrame:Lerp(look, 0.3)
    end

    if Farm.state == "approach" then
        local behind, dest
        if Config.Farm.Mode == "Mount" then
            behind = tRoot.CFrame.LookVector * -Config.Farm.MountDistance
            dest = tRoot.Position + behind + Vector3.new(0, Config.Farm.MountHeight, 0)
        elseif Config.Farm.Mode == "Orbit" then
            Farm.orbitAngle = Farm.orbitAngle + dt * 3
            local ox = math.cos(Farm.orbitAngle) * Config.Farm.MountDistance
            local oz = math.sin(Farm.orbitAngle) * Config.Farm.MountDistance
            dest = tRoot.Position + Vector3.new(ox, Config.Farm.MountHeight, oz)
        else -- Snipe
            dest = tRoot.Position + Vector3.new(0, Config.Farm.MountHeight + 3, 0)
        end
        if flyStep(myRoot, dest, Config.Farm.Speed, dt) then
            Farm.state = "mount"
            Farm.timer = 0
        end
    elseif Farm.state == "mount" then
        -- Hold position
        local behind = tRoot.CFrame.LookVector * -Config.Farm.MountDistance
        local lock = CFrame.new(tRoot.Position + behind + Vector3.new(0, Config.Farm.MountHeight, 0), tRoot.Position)
        myRoot.CFrame = lock
        Farm.timer = Farm.timer + dt
        if Farm.timer >= Config.Farm.KillDelay then
            Farm.state = "attack"
        end
    elseif Farm.state == "attack" then
        -- Aim camera at head with silent aim active → fire
        local aimPos = tHead.Position + tRoot.AssemblyLinearVelocity * 0.1
        Camera.CFrame = CFrame.new(Camera.CFrame.Position, aimPos)
        if fireWeapon() then
            Farm.shots = Farm.shots + 1
        end
        if Farm.shots >= Config.Farm.HitsPerTarget then
            Farm.shots = 0
            Farm.state = "retarget"
            Farm.target = nil
        end
    end
end)

    end

local function removeESP(p)
    local d = espCache[p]
    if d then d.bb:Destroy() end
    espCache[p] = nil
end

RunService.RenderStepped:Connect(function()
    local th = T()
    for p, d in pairs(espCache) do
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not (Config.ESP.Enabled and hum and hum.Health > 0 and root) then
            d.bb.Enabled = false
            continue
        end
        local dist3 = (Camera.CFrame.Position - root.Position).Magnitude
        if dist3 > Config.ESP.MaxDistance then
            d.bb.Enabled = false
            continue
        end
        d.bb.Adornee = root
        d.bb.Enabled = true

        local team = Config.ESP.TeamCheck and Util.team(p)
        local col = team and th.danger or th.accent

        d.stroke.Color = col
        d.name.TextColor3 = col
        d.dist.TextColor3 = col
        d.headDot.BackgroundColor3 = col

        d.name.Text = p.Name
        d.name.Visible = Config.ESP.Name
        d.box.Visible = Config.ESP.Box
        d.stroke.Enabled = Config.ESP.Box

        if Config.ESP.Distance then
            d.dist.Text = string.format("[%d]", math.floor(dist3))
            d.dist.Visible = true
        else
            d.dist.Visible = false
        end

        if Config.ESP.Health then
            local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
            d.hpBg.Visible = true
            d.hpFill.Size = UDim2.new(1, 0, hp, 0)
            d.hpFill.Position = UDim2.new(0, 0, 1 - hp, 0)
            d.hpFill.BackgroundColor3 = Color3.fromRGB(255 * (1-hp), 255 * hp, 0)
        else
            d.hpBg.Visible = false
        end

        d.headDot.Visible = Config.ESP.HeadDot
    end
end)

local function toggle(page, label, getter, setter)
    local btn = Instance.new("TextButton", page)
    btn.Size = UDim2.new(1, 0, 0, 30)
    btn.BackgroundColor3 = T().panel
    btn.TextColor3 = getter() and T().accent or T().sub
    btn.Text = (getter() and "◉  " or "○  ") .. label
    btn.Font = Enum.Font.Gotham
    btn.TextSize = 12
    btn.BorderSizePixel = 0
    local c = Instance.new("UICorner", btn)
    c.CornerRadius = UDim.new(0, 6)
    btn.MouseButton1Click:Connect(function()
        setter(not getter())
        btn.Text = (getter() and "◉  " or "○  ") .. label
        btn.TextColor3 = getter() and T().accent or T().sub
    end)
    return btn
end

local function cycle(page, label, getter, setter, options)
    local btn = Instance.new("TextButton", page)
    btn.Size = UDim2.new(1, 0, 0, 30)
    btn.BackgroundColor3 = T().panel
    btn.TextColor3 = T().text
    btn.Text = "▸ " .. label .. ": " .. tostring(getter())
    btn.Font = Enum.Font.Gotham
    btn.TextSize = 12
    btn.BorderSizePixel = 0
    local c = Instance.new("UICorner", btn)
    c.CornerRadius = UDim.new(0, 6)
    btn.MouseButton1Click:Connect(function()
        local cur = getter()
        local idx = 1
        for i, v in ipairs(options) do if v == cur then idx = i end end
        idx = (idx % #options) + 1
        setter(options[idx])
        btn.Text = "▸ " .. label .. ": " .. tostring(getter())
    end)
    return btn
end

local function slider(page, label, getter, setter, min, max, step)
    local row = Instance.new("Frame", page)
    row.Size = UDim2.new(1, 0, 0, 40)
    row.BackgroundColor3 = T().panel
    row.BorderSizePixel = 0
    local rc = Instance.new("UICorner", row)
    rc.CornerRadius = UDim.new(0, 6)

    local lbl = Instance.new("TextLabel", row)
    lbl.Size = UDim2.new(0.7, 0, 0, 20)
    lbl.Position = UDim2.new(0, 10, 0, 2)
    lbl.BackgroundTransparency = 1
    lbl.Text = label .. ": " .. tostring(getter())
    lbl.TextColor3 = T().text
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left

    local bar = Instance.new("Frame", row)
    bar.Size = UDim2.new(0.9, 0, 0, 6)
    bar.Position = UDim2.new(0.05, 0, 1, -12)
    bar.BackgroundColor3 = Color3.fromRGB(40,40,50)
    bar.BorderSizePixel = 0
    local bc = Instance.new("UICorner", bar)
    bc.CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame", bar)
    fill.BackgroundColor3 = T().accent
    fill.BorderSizePixel = 0
    fill.Size = UDim2.new((getter() - min) / (max - min), 0, 1, 0)
    local fc = Instance.new("UICorner", fill)
    fc.CornerRadius = UDim.new(1, 0)

    local function update(v)
        fill.Size = UDim2.new((v - min) / (max - min), 0, 1, 0)
        lbl.Text = label .. ": " .. tostring(v)
        setter(v)
    end

    -- Click-to-set
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
           or input.UserInputType == Enum.UserInputType.Touch then
            local conn
            conn = UIS.InputChanged:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseMovement
                   or i.UserInputType == Enum.UserInputType.Touch then
                    local mouseX = i.Position.X
                    local barAbs = bar.AbsolutePosition.X
                    local barW = bar.AbsoluteSize.X
                    local pct = math.clamp((mouseX - barAbs) / barW, 0, 1)
                    local v = min + (max - min) * pct
                    v = math.floor(v / step + 0.5) * step
                    v = math.clamp(v, min, max)
                    update(v)
                end
            end)
            local endConn
            endConn = UIS.InputEnded:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1
                   or i.UserInputType == Enum.UserInputType.Touch then
                    if conn then conn:Disconnect() end
                    if endConn then endConn:Disconnect() end
                end
            end)
        end
    end)

    return row
end

local dragging, dragStart, startPos
header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
       or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
    end
end)
UIS.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                     or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
                                   startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
       or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
