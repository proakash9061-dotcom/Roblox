local EPIX = {}
_G.EPIX = EPIX
EPIX.Chunks = {}
EPIX.Order  = {}
EPIX.Meta = {
	Name    = "Epix Backfire",
	Version = "v1",
	Credit  = "free bottom",
	Author  = "VEER",
	Build   = "499-tier",
}

function EPIX.register(n, label, fn)
	EPIX.Chunks[n] = { fn = fn, label = label }
	table.insert(EPIX.Order, n)
	print(("[EPIX] chunk %d loaded -> %s"):format(n, label))
end

function EPIX.run()
	table.sort(EPIX.Order)
	local missing = {}
	for i = 1, EPIX.Order[#EPIX.Order] do
		if not EPIX.Chunks[i] then table.insert(missing, i) end
	end
	if #missing > 0 then
		warn("[EPIX] MISSING CHUNKS: " .. table.concat(missing, ", "))
		return
	end
	for _, n in ipairs(EPIX.Order) do
		local c = EPIX.Chunks[n]
		local ok, err = pcall(c.fn)
		if not ok then
			warn(("[EPIX] chunk %d (%s) ERROR: %s"):format(n, c.label, tostring(err)))
		end
	end
	print(("[EPIX] %s %s online  ·  credit: %s")
		:format(EPIX.Meta.Name, EPIX.Meta.Version, EPIX.Meta.Credit))
end

EPIX.register(3, "entities", function()
	local Players    = EPIX.S.Players
	local Workspace  = EPIX.S.Workspace
	local LP         = EPIX.LP
	local Camera     = EPIX.Camera

	local Entity = {}

	function Entity.getChar(p)
		return p and p.Character
	end

	function Entity.getHum(p)
		local c = Entity.getChar(p)
		return c and c:FindFirstChildOfClass("Humanoid")
	end

	function Entity.getRoot(p)
		local c = Entity.getChar(p)
		return c and c:FindFirstChild("HumanoidRootPart")
	end

	function Entity.getHead(p)
		local c = Entity.getChar(p)
		return c and c:FindFirstChild("Head")
	end

	function Entity.getTorso(p)
		local c = Entity.getChar(p)
		return c and (c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso"))
	end

	function Entity.isAlive(p)
		local hum = Entity.getHum(p)
		return hum and hum.Health > 0
	end

	function Entity.isFriendly(p, teamCheck)
		if not teamCheck then return false end
		return p.Team ~= nil and p.Team == LP.Team
	end

	function Entity.distTo(p)
		local root = Entity.getRoot(LP)
		local hr   = Entity.getRoot(p)
		if not root or not hr then return math.huge end
		return (root.Position - hr.Position).Magnitude
	end

	function Entity.isVisible(p, targetPart)
		if not targetPart then return false end
		local cam = Workspace.CurrentCamera
		if not cam then return false end
		local origin = cam.CFrame.Position
		local dir = (targetPart.Position - origin)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { LP.Character }
		local hit = Workspace:Raycast(origin, dir, params)
		return hit == nil or hit.Instance:IsDescendantOf(p.Character)
	end

	function Entity.getBones(p)
		local char = Entity.getChar(p)
		if not char then return nil end
		local head = char:FindFirstChild("Head")
		local hrp  = char:FindFirstChild("HumanoidRootPart")
		if not head or not hrp then return nil end
		local hum = Entity.getHum(p)
		local rig = hum and hum.RigType
		local bones = {
			Head = head,
			Root = hrp,
		}
		if rig == Enum.HumanoidRigType.R15 then
			bones.UpperTorso = char:FindFirstChild("UpperTorso")
			bones.LowerTorso = char:FindFirstChild("LowerTorso")
			bones.LeftUpperArm = char:FindFirstChild("LeftUpperArm")
			bones.LeftLowerArm = char:FindFirstChild("LeftLowerArm")
			bones.LeftHand = char:FindFirstChild("LeftHand")
			bones.RightUpperArm = char:FindFirstChild("RightUpperArm")
			bones.RightLowerArm = char:FindFirstChild("RightLowerArm")
			bones.RightHand = char:FindFirstChild("RightHand")
			bones.LeftUpperLeg = char:FindFirstChild("LeftUpperLeg")
			bones.LeftLowerLeg = char:FindFirstChild("LeftLowerLeg")
			bones.LeftFoot = char:FindFirstChild("LeftFoot")
			bones.RightUpperLeg = char:FindFirstChild("RightUpperLeg")
			bones.RightLowerLeg = char:FindFirstChild("RightLowerLeg")
			bones.RightFoot = char:FindFirstChild("RightFoot")
		else
			bones.Torso = char:FindFirstChild("Torso")
			bones.LeftArm = char:FindFirstChild("Left Arm")
			bones.RightArm = char:FindFirstChild("Right Arm")
			bones.LeftLeg = char:FindFirstChild("Left Leg")
			bones.RightLeg = char:FindFirstChild("Right Leg")
		end
		return bones
	end

	function Entity.getPlayersList()
		local list = {}
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP and Entity.isAlive(p) then
				table.insert(list, p)
			end
		end
		return list
	end

	function Entity.worldToScreen(worldPos)
		local cam = Workspace.CurrentCamera
		if not cam then return nil end
		local sp, onScreen = cam:WorldToViewportPoint(worldPos)
		if not onScreen or sp.Z <= 0 then return nil end
		return Vector2.new(sp.X, sp.Y), sp.Z
	end

	function Entity.getScreenBounds(p)
		local head = Entity.getHead(p)
		local root = Entity.getRoot(p)
		if not head or not root then return nil end
		local headPos = head.Position + Vector3.new(0, 0.6, 0)
		local feetPos = root.Position - Vector3.new(0, 3, 0)
		local sHead = Entity.worldToScreen(headPos)
		local sFeet = Entity.worldToScreen(feetPos)
		if not sHead or not sFeet then return nil end
		local top    = sHead
		local bottom = sFeet
		local height = math.abs(bottom.Y - top.Y)
		local width  = height * 0.55
		local left   = top.X - width / 2
		local right  = top.X + width / 2
		return {
			topLeft = Vector2.new(left, top.Y),
			bottomRight = Vector2.new(right, bottom.Y),
			center = Vector2.new((left + right) / 2, (top.Y + bottom.Y) / 2),
			width = width,
			height = height,
		}
	end

	EPIX.Entity = Entity
	print("[EPIX] entity helpers ready")
end)

		local b = EPIX.State.ESP_Registry[p]
		if b and b.container and b.container.Parent then return b end
		b = createBundle(p)
		EPIX.State.ESP_Registry[p] = b
		return b
	end

	local function clearBundle(p)
		local b = EPIX.State.ESP_Registry[p]
		if b and b.container then
			pcall(function() b.container:Destroy() end)
		end
		EPIX.State.ESP_Registry[p] = nil
	end

	-- outline side sizing
	local function sizeOutline(outline, w, h, thickness)
		local t = thickness or 1
		local sides = outline:GetChildren()
		for _, s in ipairs(sides) do
			if s:IsA("Frame") then
				if s.Name == "Top" or s.Name == "Bottom" then
					s.Size = UDim2.new(1, 0, 0, t)
					s.Position = (s.Name == "Top") and UDim2.new(0, 0, 0, 0) or UDim2.new(0, 0, 1, -t)
				elseif s.Name == "Left" or s.Name == "Right" then
					s.Size = UDim2.new(0, t, 1, 0)
					s.Position = (s.Name == "Left") and UDim2.new(0, 0, 0, 0) or UDim2.new(1, -t, 0, 0)
				end
			end
		end
	end

	-- corner-style box sizing
	local function sizeCorners(box, w, h, thickness)
		local t = thickness or 1
		local cl = math.max(4, math.floor(math.min(w, h) * 0.25))
		for _, s in ipairs(box:GetChildren()) do
			if s:IsA("Frame") then
				if s.Name == "Top" then
					s.Size = UDim2.new(0, cl, 0, t)
					s.Position = UDim2.new(0, 0, 0, 0)
				elseif s.Name == "Bottom" then
					s.Size = UDim2.new(0, cl, 0, t)
					s.Position = UDim2.new(1, -cl, 1, -t)
				elseif s.Name == "Left" then
					s.Size = UDim2.new(0, t, 0, cl)
					s.Position = UDim2.new(0, 0, 1, -cl)
				elseif s.Name == "Right" then
					s.Size = UDim2.new(0, t, 0, cl)
					s.Position = UDim2.new(1, -t, 0, 0)
				end
			end
		end
	end

	local function sizeFullBox(box, w, h, thickness)
		local t = thickness or 1
		for _, s in ipairs(box:GetChildren()) do
			if s:IsA("Frame") then
				if s.Name == "Top" or s.Name == "Bottom" then
					s.Size = UDim2.new(1, 0, 0, t)
					s.Position = (s.Name == "Top") and UDim2.new(0, 0, 0, 0) or UDim2.new(0, 0, 1, -t)
				elseif s.Name == "Left" or s.Name == "Right" then
					s.Size = UDim2.new(0, t, 1, 0)
					s.Position = (s.Name == "Left") and UDim2.new(0, 0, 0, 0) or UDim2.new(1, -t, 0, 0)
				end
			end
		end
	end

	-- skeleton drawing
	local BONE_EDGES = {
		-- head-neck-torso chain
		{"Head", "UpperTorso"},
		{"UpperTorso", "LowerTorso"},
		-- arms
		{"UpperTorso", "LeftUpperArm"},
		{"LeftUpperArm", "LeftLowerArm"},
		{"LeftLowerArm", "LeftHand"},
		{"UpperTorso", "RightUpperArm"},
		{"RightUpperArm", "RightLowerArm"},
		{"RightLowerArm", "RightHand"},
		-- legs
		{"LowerTorso", "LeftUpperLeg"},
		{"LeftUpperLeg", "LeftLowerLeg"},
		{"LeftLowerLeg", "LeftFoot"},
		{"LowerTorso", "RightUpperLeg"},
		{"RightUpperLeg", "RightLowerLeg"},
		{"RightLowerLeg", "RightFoot"},
	}

	local function drawSkeleton(skelFrame, bones)
		if not bones then return end
		local used = 0

		for _, edge in ipairs(BONE_EDGES) do
			local a = bones[edge[1]]
			local b = bones[edge[2]]
			if a and b then
				local aPos = Entity.worldToScreen(a.Position)
				local bPos = Entity.worldToScreen(b.Position)
				if aPos and bPos then
					used = used + 1
					local line = skelFrame:FindFirstChild("L" .. used)
					if not line then
						line = Instance.new("Frame")
						line.Name = "L" .. used
						line.BackgroundColor3 = CONFIG.ESP.BoxColor
						line.BorderSizePixel = 0
						line.ZIndex = 6
						line.Parent = skelFrame
					end
					local dx = bPos.X - aPos.X
					local dy = bPos.Y - aPos.Y
					local len = math.sqrt(dx * dx + dy * dy)
					local angle = math.deg(math.atan2(dy, dx))
					line.Visible = true
					line.Size = UDim2.new(0, len, 0, 1)
					line.Position = UDim2.new(0, aPos.X, 0, aPos.Y)
					line.Rotation = angle
					line.AnchorPoint = Vector2.new(0, 0.5)
				end
			end
		end

		-- hide unused lines
		for _, c in ipairs(skelFrame:GetChildren()) do
			if c:IsA("Frame") then
				local idx = tonumber(c.Name:match("L(%d+)"))
				if idx and idx > used then c.Visible = false end
			end
		end
	end

	-- render loop
	local function renderOne(p)
		local b = ensureBundle(p)
		local bounds = Entity.getScreenBounds(p)
		if not bounds then
			b.container.Visible = false
			return
		end

		local hum = Entity.getHum(p)
		if not hum then b.container.Visible = false return end

		b.container.Visible = true
		b.container.Position = UDim2.fromOffset(bounds.topLeft.X, bounds.topLeft.Y)
		b.container.Size = UDim2.fromOffset(bounds.width, bounds.height)

		-- outline
		b.outline.Visible = CONFIG.ESP.Outline
		if CONFIG.ESP.Outline then
			b.outline.Size = UDim2.new(1, 4, 1, 4)
			b.outline.Position = UDim2.new(0, -2, 0, -2)
			sizeOutline(b.outline, bounds.width + 4, bounds.height + 4, 1)
		end

		-- box
		b.box.Visible = CONFIG.ESP.Box
		if CONFIG.ESP.Box then
			b.box.Size = UDim2.new(1, 0, 1, 0)
			b.box.Position = UDim2.new(0, 0, 0, 0)
			if CONFIG.ESP.BoxStyle == "Corner" then
				sizeCorners(b.box, bounds.width, bounds.height, 2)
			else
				sizeFullBox(b.box, bounds.width, bounds.height, 1.5)
			end
		end

		-- fill
		b.fill.Visible = CONFIG.ESP.Chams
		b.fill.Size = UDim2.new(1, 0, 1, 0)

		-- name
		b.name.Visible = CONFIG.ESP.Name
		if CONFIG.ESP.Name then
			b.name.Text = p.Name
			b.name.Size = UDim2.new(1, 0, 0, 14)
			b.name.Position = UDim2.new(0, 0, 0, -16)
		end

		-- distance
		b.dist.Visible = CONFIG.ESP.Distance
		if CONFIG.ESP.Distance then
			local d = Entity.distTo(p)
			b.dist.Text = string.format("%dm", math.floor(d))
			b.dist.Size = UDim2.new(1, 0, 0, 12)
			b.dist.Position = UDim2.new(0, 0, 1, 2)
		end

		-- health
		b.hpBack.Visible = CONFIG.ESP.HealthBar
		b.hpFill.Visible = CONFIG.ESP.HealthBar
		if CONFIG.ESP.HealthBar then
			local hpRatio = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
			b.hpBack.Size = UDim2.new(0, 3, 1, 0)
			b.hpBack.Position = UDim2.new(0, -6, 0, 0)
			b.hpFill.Size = UDim2.new(1, 0, hpRatio, 0)
			b.hpFill.Position = UDim2.new(0, 0, 1 - hpRatio, 0)
			local r = 1 - hpRatio
			b.hpFill.BackgroundColor3 = Color3.new(
				math.clamp(r * 2, 0, 1),
				math.clamp((1 - r) * 2, 0, 1),
				0.2
			)
		end

		-- skeleton
		b.skeleton.Visible = CONFIG.ESP.Skeleton
		if CONFIG.ESP.Skeleton then
			b.skeleton.Size = UDim2.new(1, 0, 1, 0)
			b.skeleton.Position = UDim2.new(0, 0, 0, 0)
			drawSkeleton(b.skeleton, Entity.getBones(p))
		end
	end

	function ESP.start()
		if EPIX.State.ESP_Conn then
			EPIX.State.ESP_Conn:Disconnect()
		end
		EPIX.State.ESP_Conn = RunService.RenderStepped:Connect(function()
			if not CONFIG.ESP.Enabled then
				for _, b in next, EPIX.State.ESP_Registry do
					if b.container then b.container.Visible = false end
				end
				return
			end
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= LP and Entity.isAlive(p)
				and not Entity.isFriendly(p, CONFIG.ESP.TeamCheck)
				and Entity.distTo(p) <= CONFIG.ESP.MaxDist then
					pcall(renderOne, p)
				else
					local b = EPIX.State.ESP_Registry[p]
					if b and b.container then b.container.Visible = false end
				end
			end
		end)
	end

	function ESP.stop()
		if EPIX.State.ESP_Conn then
			EPIX.State.ESP_Conn:Disconnect()
			EPIX.State.ESP_Conn = nil
		end
		for p, b in next, EPIX.State.ESP_Registry do
			if b.container then b.container.Visible = false end
		end
	end

	function ESP.refreshStyle()
		for _, b in next, EPIX.State.ESP_Registry do
			for _, s in ipairs(b.box:GetChildren()) do
				if s:IsA("Frame") then s.BackgroundColor3 = CONFIG.ESP.BoxColor end
			end
			for _, s in ipairs(b.outline:GetChildren()) do
				if s:IsA("Frame") then s.BackgroundColor3 = CONFIG.ESP.OutlineColor end
			end
			b.name.TextColor3 = CONFIG.ESP.BoxColor
			b.dist.TextColor3 = CONFIG.ESP.BoxColor
			b.fill.BackgroundColor3 = CONFIG.ESP.FilledColor
			b.fill.BackgroundTransparency = 1 - CONFIG.ESP.FilledAlpha
			b.name.TextSize = CONFIG.ESP.TextSize
			b.dist.TextSize = CONFIG.ESP.TextSize - 2
		end
	end

	function ESP.clearAll()
		for p, _ in next, EPIX.State.ESP_Registry do
			clearBundle(p)
		end
	end

	EPIX.ESP = ESP
	print("[EPIX] ESP engine ready")
end)

EPIX.register(5, "aimbot", function()
	local RunService = EPIX.S.RunService
	local Workspace  = EPIX.S.Workspace
	local Players    = EPIX.S.Players
	local UIS        = EPIX.S.UserInputService

	local Entity     = EPIX.Entity
	local LP         = EPIX.LP
	local CONFIG     = EPIX.CONFIG
	local T          = EPIX.T

	local Aimbot = {}

	-- FOV ring drawn on screen
	local FOVRing
	local function ensureFOVRing()
		if FOVRing and FOVRing.Parent then return FOVRing end
		local ring = Instance.new("Frame")
		ring.Name = "EpixFOVRing"
		ring.AnchorPoint = Vector2.new(0.5, 0.5)
		ring.Position = UDim2.new(0.5, 0, 0.5, 0)
		ring.BackgroundTransparency = 1
		ring.BorderSizePixel = 0
		ring.ZIndex = 50
		ring.Visible = false

		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(1, 0)
		corner.Parent = ring

		local stroke = Instance.new("UIStroke")
		stroke.Color = CONFIG.Theme.Accent
		stroke.Thickness = 1.4
		stroke.Transparency = 0.35
		stroke.Parent = ring

		ring.Parent = EPIX.PARENT
		FOVRing = ring
		return ring
	end

	local function updateFOVRing()
		local ring = ensureFOVRing()
		if not CONFIG.Aimbot.Enabled and not CONFIG.Silent.Enabled then
			ring.Visible = false
			return
		end
		ring.Visible = true
		local fov = math.max(CONFIG.Aimbot.FOV, CONFIG.Silent.FOV)
		local size = fov * 2
		ring.Size = UDim2.fromOffset(size, size)
		local stroke = ring:FindFirstChildOfClass("UIStroke")
		if stroke then
			stroke.Color = CONFIG.Silent.Enabled and Color3.fromRGB(255, 90, 180) or CONFIG.Theme.Accent
		end
	end

	-- target selection
	local function isBehindWall(fromPos, part)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { LP.Character }
		local dir = part.Position - fromPos
		local hit = Workspace:Raycast(fromPos, dir, params)
		return hit ~= nil and not hit.Instance:IsDescendantOf(part.Parent)
	end

	local function getHitPart(p, mode)
		if mode == "Head" then return Entity.getHead(p) end
		if mode == "Torso" then return Entity.getTorso(p) end
		if mode == "HeadTorso" then
			return Entity.getHead(p) or Entity.getTorso(p)
		end
		return Entity.getHead(p)
	end

	function Aimbot.findTarget(opts)
		opts = opts or {}
		local cam = Workspace.CurrentCamera
		if not cam then return nil end
		local myPos = Entity.getRoot(LP) and Entity.getRoot(LP).Position or cam.CFrame.Position
		local best, bestScore = nil, math.huge
		local fov = opts.FOV or CONFIG.Aimbot.FOV
		local maxDist = opts.MaxDist or CONFIG.Aimbot.MaxDist
		local teamCheck = opts.TeamCheck ~= false
		local wallCheck = opts.WallCheck ~= false
		local hitMode = opts.HitPart or CONFIG.Aimbot.HitPart
		local screenCenter = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP and Entity.isAlive(p) then
				if not Entity.isFriendly(p, teamCheck) then
					local dist = Entity.distTo(p)
					if dist <= maxDist then
						local part = getHitPart(p, hitMode)
						if part then
							local sp = Entity.worldToScreen(part.Position)
							if sp then
								local mag = (sp - screenCenter).Magnitude
								if mag <= fov then
									if not wallCheck or Entity.isVisible(p, part) then
										-- score: prefer closer to crosshair, then closer to player
										local score = mag + dist * 0.05
										if score < bestScore then
											bestScore = score
											best = { player = p, part = part, screenPos = sp, dist = dist }
										end
									end
								end
							end
						end
					end
				end
			end
		end
		return best
	end

	-- apply aim
	local function aimAt(cam, target, smooth)
		local goalCF = CFrame.new(cam.CFrame.Position, target.part.Position)
		if smooth >= 1 then
			cam.CFrame = goalCF
		else
			cam.CFrame = cam.CFrame:Lerp(goalCF, smooth)
		end
	end

	-- lock state
	local aimActive = false
	local aimKeyHeld = false
	UIS.InputBegan:Connect(function(i, gp)
		if gp then return end
		if i.KeyCode == CONFIG.Aimbot.Key then aimKeyHeld = true end
		if i.KeyCode == CONFIG.Keybind or i.KeyCode == CONFIG.KeybindAlt then
			EPIX.State.GuiRefs.Toggle and EPIX.State.GuiRefs.Toggle()
		end
	end)
	UIS.InputEnded:Connect(function(i)
		if i.KeyCode == CONFIG.Aimbot.Key then aimKeyHeld = false end
	end)

	-- main loop
	function Aimbot.start()
		if EPIX.State.Aimbot_Conn then EPIX.State.Aimbot_Conn:Disconnect() end
		EPIX.State.Aimbot_Conn = RunService.RenderStepped:Connect(function(dt)
			updateFOVRing()
			if not CONFIG.Aimbot.Enabled then return end
			local cam = Workspace.CurrentCamera
			if not cam then return end

			-- cam tracking (auto lock regardless of key)
			if CONFIG.Aimbot.CamTrack then
				local tgt = Aimbot.findTarget({
					FOV = CONFIG.Aimbot.FOV,
					MaxDist = CONFIG.Aimbot.MaxDist,
					TeamCheck = CONFIG.Aimbot.TeamCheck,
					WallCheck = CONFIG.Aimbot.WallCheck,
					HitPart = CONFIG.Aimbot.HitPart,
				})
				EPIX.State.Aimbot_Target = tgt
				if tgt then aimAt(cam, tgt, CONFIG.Aimbot.CamSmooth) end
			end

			-- key-held aim
			if aimKeyHeld or CONFIG.Aimbot.AlwaysOn then
				local tgt = Aimbot.findTarget({
					FOV = CONFIG.Aimbot.FOV,
					MaxDist = CONFIG.Aimbot.MaxDist,
					TeamCheck = CONFIG.Aimbot.TeamCheck,
					WallCheck = CONFIG.Aimbot.WallCheck,
					HitPart = CONFIG.Aimbot.HitPart,
				})
				EPIX.State.Aimbot_Target = tgt
				if tgt then
					aimAt(cam, tgt, CONFIG.Aimbot.Smoothing)
					-- autofire
					if CONFIG.Aimbot.AutoFire and tick() - (EPIX.State.LastAutoFire or 0) > CONFIG.Aimbot.AutoFireRate then
						EPIX.State.LastAutoFire = tick()
						pcall(function()
							if mouse1click then mouse1click() end
							if mouse1press and mouse1release then
								mouse1press(); task.wait(0.02); mouse1release()
							end
						end)
					end
				end
			end
		end)
	end

	function Aimbot.stop()
		if EPIX.State.Aimbot_Conn then
			EPIX.State.Aimbot_Conn:Disconnect()
			EPIX.State.Aimbot_Conn = nil
		end
		if FOVRing then FOVRing.Visible = false end
	end

	function Aimbot.target()
		return EPIX.State.Aimbot_Target
	end

	EPIX.Aimbot = Aimbot
	print("[EPIX] aimbot engine ready")
end)

EPIX.register(6, "silent", function()
	local RunService = EPIX.S.RunService
	local Workspace  = EPIX.S.Workspace
	local Players    = EPIX.S.Players
	local UIS        = EPIX.S.UserInputService

	local Entity     = EPIX.Entity
	local LP         = EPIX.LP
	local CONFIG     = EPIX.CONFIG

	local Silent = {}
	local Aimbot  = EPIX.Aimbot

	-- Silent aim: hook the raycast / namecall of the game's gun client.
	-- Falls back to spoofing the mouse hit if game uses UserInputService.

	local hookInstalled = false
	local oldNamecall   = nil

	local function getSilentTarget()
		return Aimbot.findTarget({
			FOV = CONFIG.Silent.FOV,
			MaxDist = CONFIG.Silent.MaxDist,
			TeamCheck = CONFIG.Silent.TeamCheck,
			WallCheck = CONFIG.Silent.WallCheck,
			HitPart = CONFIG.Silent.HitPart,
		})
	end

	local function rewriteArg(arg)
		if typeof(arg) == "Vector3" then
			local tgt = EPIX.State.Silent_Target
			if tgt then
				local predicted = tgt.part.Position + (tgt.part.AssemblyLinearVelocity or Vector3.zero) * CONFIG.Silent.PredAmt
				return predicted
			end
		end
		if typeof(arg) == "CFrame" then
			local tgt = EPIX.State.Silent_Target
			if tgt then
				return CFrame.new(tgt.part.Position)
			end
		end
		if typeof(arg) == "Instance" and arg:IsA("BasePart") then
			local tgt = EPIX.State.Silent_Target
			if tgt then
				return tgt.part
			end
		end
		return arg
	end

	function Silent.install()
		if hookInstalled then return true end
		local mt = getrawmetatable and getrawmetatable(game)
		if not mt then return false, "no getrawmetatable" end

		local oldNC = mt.__namecall
		oldNamecall = oldNC
		setreadonly(mt, false)

		mt.__namecall = newcclosure(function(self, ...)
			local method = getnamecallmethod()
			if CONFIG.Silent.Enabled and not checkcaller() then
				local tgt = getSilentTarget()
				EPIX.State.Silent_Target = tgt
				if tgt then
					local args = { ... }
					local newArgs = {}
					for i = 1, #args do
						newArgs[i] = rewriteArg(args[i])
					end
					return oldNC(self, unpack(newArgs))
				end
			end
			return oldNC(self, ...)
		end)

		setreadonly(mt, true)
		hookInstalled = true
		return true
	end

	function Silent.uninstall()
		if not hookInstalled then return end
		local mt = getrawmetatable and getrawmetatable(game)
		if mt and oldNamecall then
			setreadonly(mt, false)
			mt.__namecall = oldNamecall
			setreadonly(mt, true)
		end
		hookInstalled = false
	end

	-- optional: UserInputService MouseClick rewrite (some games)
	local function installMouseHook()
		if EPIX.State.Silent_MouseConn then return end
		local oldBind = UIS.InputBegan
		-- no-op — the namecall hook covers most cases
	end

	function Silent.start()
		Silent.install()
		installMouseHook()
		-- heartbeat refresher so target is fresh even without a click
		if EPIX.State.Silent_Conn then EPIX.State.Silent_Conn:Disconnect() end
		EPIX.State.Silent_Conn = RunService.Heartbeat:Connect(function()
			if CONFIG.Silent.Enabled then
				EPIX.State.Silent_Target = getSilentTarget()
			else
				EPIX.State.Silent_Target = nil
			end
		end)
	end

	function Silent.stop()
		if EPIX.State.Silent_Conn then
			EPIX.State.Silent_Conn:Disconnect()
			EPIX.State.Silent_Conn = nil
		end
		EPIX.State.Silent_Target = nil
	end

	EPIX.Silent = Silent
	print("[EPIX] silent aim module ready")
end)

EPIX.register(7, "autofarm", function()
	local RunService = EPIX.S.RunService
	local Workspace  = EPIX.S.Workspace
	local Players    = EPIX.S.Players

	local Entity     = EPIX.Entity
	local LP         = EPIX.LP
	local CONFIG     = EPIX.CONFIG

	local Autofarm = {}

	local function pickNextTarget(exclude)
		local best, bestD = nil, math.huge
		local myRoot = Entity.getRoot(LP)
		if not myRoot then return nil end
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= LP and p ~= exclude and Entity.isAlive(p) then
				if not Entity.isFriendly(p, CONFIG.Autofarm.TeamCheck) then
					local d = Entity.distTo(p)
					if d <= CONFIG.Autofarm.Range and d < bestD then
						bestD = d
						best = p
					end
				end
			end
		end
		return best
	end

	local function rideOn(target)
		local myRoot = Entity.getRoot(LP)
		local tgtRoot = Entity.getRoot(target)
		if not myRoot or not tgtRoot then return false end
		local tgtHead = Entity.getHead(target)
		local anchor = tgtHead and tgtHead.Position or tgtRoot.Position
		myRoot.CFrame = CFrame.new(anchor + Vector3.new(0, 2.5, 0))
		return true
	end

	local function shootHead(target)
		local head = Entity.getHead(target)
		if not head then return false end
		local cam = Workspace.CurrentCamera
		if not cam then return false end
		cam.CFrame = CFrame.new(cam.CFrame.Position, head.Position)
		task.wait(0.05)
		pcall(function()
			if mouse1click then mouse1click() end
			if mouse1press and mouse1release then
				mouse1press(); task.wait(0.02); mouse1release()
			end
		end)
		return true
	end

	local function runCycle()
		if not CONFIG.Autofarm.Enabled then return end
		local target = pickNextTarget(EPIX.State.Autofarm_Current)
		if not target then
			task.wait(0.4)
			return
		end
		EPIX.State.Autofarm_Current = target

		-- ride
		for _ = 1, 3 do
			if not rideOn(target) then break end
			task.wait(CONFIG.Autofarm.HopDelay)
		end

		-- headshot
		for _ = 1, 3 do
			if not Entity.isAlive(target) then break end
			shootHead(target)
			task.wait(CONFIG.Autofarm.KillDelay)
		end

		EPIX.State.Autofarm_Cycles = (EPIX.State.Autofarm_Cycles or 0) + 1
		if CONFIG.Autofarm.MaxCycles > 0
		and EPIX.State.Autofarm_Cycles >= CONFIG.Autofarm.MaxCycles then
			CONFIG.Autofarm.Enabled = false
			EPIX.State.Autofarm_Active = false
		end
	end

	function Autofarm.start()
		if EPIX.State.Autofarm_Conn then EPIX.State.Autofarm_Conn:Disconnect() end
		EPIX.State.Autofarm_Active = true
		EPIX.State.Autofarm_Cycles = 0
		EPIX.State.Autofarm_Current = nil
		EPIX.State.Autofarm_Conn = RunService.Heartbeat:Connect(function()
			if not CONFIG.Autofarm.Enabled then return end
			if EPIX.State.Autofarm_Busy then return end
			EPIX.State.Autofarm_Busy = true
			task.spawn(function()
				local ok, err = pcall(runCycle)
				if not ok then warn("[EPIX autofarm]", err) end
				EPIX.State.Autofarm_Busy = false
			end)
		end)
	end

	function Autofarm.stop()
		if EPIX.State.Autofarm_Conn then
			EPIX.State.Autofarm_Conn:Disconnect()
			EPIX.State.Autofarm_Conn = nil
		end
		EPIX.State.Autofarm_Active = false
	end

	EPIX.Autofarm = Autofarm
	print("[EPIX] autofarm ready")
end)

EPIX.register(8, "extras", function()
	local RunService = EPIX.S.RunService
	local Workspace  = EPIX.S.Workspace
	local Lighting   = EPIX.S.Lighting
	local Stats      = EPIX.S.Stats
	local Players    = EPIX.S.Players

	local Entity     = EPIX.Entity
	local LP         = EPIX.LP
	local CONFIG     = EPIX.CONFIG
	local T          = EPIX.T

	--========================================================================
	-- CLIENT RGB (only your own UGC / gun / knife)
	--========================================================================
	local RGB = {}

	local function allDescendants(root)
		local list = {}
		local stack = { root }
		while #stack > 0 do
			local n = table.remove(stack)
			table.insert(list, n)
			for _, c in ipairs(n:GetChildren()) do
				table.insert(stack, c)
			end
		end
		return list
	end

	local function isMine(inst)
		if not inst then return false end
		local ok, p = pcall(function() return Players:GetPlayerFromCharacter(inst) end)
		if ok and p == LP then return true end
		if inst:IsDescendantOf(LP.Character) then return true end
		return false
	end

	local function tintPart(part, color)
		if part:IsA("BasePart") then
			pcall(function()
				part.Color = color
				if part:FindFirstChildOfClass("SurfaceAppearance") then
					part:FindFirstChildOfClass("SurfaceAppearance"):Destroy()
				end
			end)
		elseif part:IsA("Decal") or part:IsA("Texture") then
			pcall(function() part.Color3 = color end)
		elseif part:IsA("SpecialMesh") or part:IsA("MeshPart") then
			pcall(function() part.VertexColor = Vector3.new(color.R, color.G, color.B) end)
		end
	end

	function RGB.start()
		if EPIX.State.RGB_Conn then EPIX.State.RGB_Conn:Disconnect() end
		EPIX.State.RGB_Conn = RunService.Heartbeat:Connect(function()
			if not CONFIG.ClientRGB.Enabled then return end
			local color = EPIX.Theme.rgb(CONFIG.ClientRGB.Speed)
			local targets = {}

			if CONFIG.ClientRGB.ApplyChar then
				if LP.Character then
					for _, d in ipairs(allDescendants(LP.Character)) do
						table.insert(targets, d)
					end
				end
			end

			if CONFIG.ClientRGB.ApplyWeapons then
				local char = LP.Character
				if char then
					for _, c in ipairs(char:GetChildren()) do
						if c:IsA("Tool") then
							for _, d in ipairs(allDescendants(c)) do
								table.insert(targets, d)
							end
						end
					end
				end
			end

			for _, d in ipairs(targets) do
				if isMine(d) then tintPart(d, color) end
			end
		end)
	end

	function RGB.stop()
		if EPIX.State.RGB_Conn then
			EPIX.State.RGB_Conn:Disconnect()
			EPIX.State.RGB_Conn = nil
		end
	end

	EPIX.RGB = RGB

	--========================================================================
	-- POTATO MODE (kill grass, trees, decals, particles)
	--========================================================================
	local Potato = {}

	local function hideInstance(inst)
		EPIX.State.Potato_Saved[inst] = inst.Parent
		pcall(function()
			if inst:IsA("BasePart") then inst.Transparency = 1 end
			if inst:IsA("Decal") then inst.Transparency = 1 end
			if inst:IsA("Texture") then inst.Transparency = 1 end
			if inst:IsA("ParticleEmitter") then inst.Enabled = false end
			if inst:IsA("Trail") then inst.Enabled = false end
			if inst:IsA("Beam") then inst.Enabled = false end
			if inst:IsA("BillboardGui") then inst.Enabled = false end
			if inst:IsA("SurfaceGui") then inst.Enabled = false end
			if inst:IsA("Model") then
				for _, d in ipairs(inst:GetDescendants()) do
					if d:IsA("BasePart") then d.Transparency = 1 end
				end
			end
		end)
	end

	function Potato.apply()
		local saved = {}
		local function sweep(container)
			for _, inst in ipairs(container:GetChildren()) do
				local n = inst.Name:lower()
				if CONFIG.Potato.RemoveGrass and (n:find("grass") or n:find("terrain") and inst:IsA("BasePart")) then
					hideInstance(inst)
				elseif CONFIG.Potato.RemoveTrees and (n:find("tree") or n:find("bush") or n:find("foliage")) then
					hideInstance(inst)
				elseif CONFIG.Potato.RemoveDecals and inst:IsA("Decal") then
					hideInstance(inst)
				elseif CONFIG.Potato.RemoveParticles and (inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Beam")) then
					hideInstance(inst)
				elseif CONFIG.Potato.RemoveBillboards and inst:IsA("BillboardGui") then
					hideInstance(inst)
				elseif inst:IsA("Folder") or inst:IsA("Model") then
					sweep(inst)
				end
			end
		end
		pcall(function() sweep(Workspace) end)

		if CONFIG.Potato.ForceLOD then
			pcall(function()
				for _, d in ipairs(Workspace:GetDescendants()) do
					if d:IsA("Model") then
						pcall(function() d.LevelOfDetail = Enum.ModelLevelOfDetail.Disabled end)
					end
				end
			end)
		end

		if CONFIG.Potato.SkipMaterials then
			pcall(function()
				Lighting.GlobalShadows = false
				Lighting.FogEnd = 5000
			end)
		end
	end

	function Potato.restore()
		for inst, parent in next, EPIX.State.Potato_Saved do
			if inst and inst.Parent == nil and parent then
				pcall(function() inst.Parent = parent end)
			end
			pcall(function()
				if inst:IsA("BasePart") then inst.Transparency = 0 end
				if inst:IsA("Decal") then inst.Transparency = 0 end
				if inst:IsA("Texture") then inst.Transparency = 0 end
				if inst:IsA("ParticleEmitter") then inst.Enabled = true end
				if inst:IsA("Trail") then inst.Enabled = true end
				if inst:IsA("Beam") then inst.Enabled = true end
				if inst:IsA("BillboardGui") then inst.Enabled = true end
			end)
		end
		EPIX.State.Potato_Saved = {}
	end

	EPIX.Potato = Potato

	--========================================================================
	-- ASSET CULLER (hide building/props beyond max dist)
	--========================================================================
	local Culler = {}

	function Culler.start()
		if EPIX.State.Culler_Conn then EPIX.State.Culler_Conn:Disconnect() end
		EPIX.State.Culler_Conn = RunService.Heartbeat:Connect(function()
			if not CONFIG.Culler.Enabled then return end
			local myRoot = Entity.getRoot(LP)
			if not myRoot then return end
			local cam = Workspace.CurrentCamera
			if not cam then return end
			for _, inst in ipairs(Workspace:GetChildren()) do
				if inst:IsA("Model") or inst:IsA("BasePart") then
					local n = inst.Name:lower()
					local isTarget = false
					if CONFIG.Culler.Targets == "All" then isTarget = true
					elseif CONFIG.Culler.Targets == "Building" and
						(n:find("build") or n:find("house") or n:find("structure") or
						 n:find("wall") or n:find("tower")) then isTarget = true
					elseif CONFIG.Culler.Targets == "Props" and
						(n:find("prop") or n:find("crate") or n:find("barrel")) then isTarget = true end
					if isTarget then
						local ok, pos = pcall(function()
							return inst:IsA("BasePart") and inst.Position or inst:GetPivot().Position
						end)
						if ok and pos then
							local d = (pos - myRoot.Position).Magnitude
							local shouldHide = d > CONFIG.Culler.MaxDist
							pcall(function()
								if shouldHide then
									if inst:IsA("BasePart") then inst.LocalTransparencyModifier = 1
									else for _, p in ipairs(inst:GetDescendants()) do
										if p:IsA("BasePart") then p.LocalTransparencyModifier = 1 end
									end end
								else
									if inst:IsA("BasePart") then inst.LocalTransparencyModifier = 0
									else for _, p in ipairs(inst:GetDescendants()) do
										if p:IsA("BasePart") then p.LocalTransparencyModifier = 0 end
									end end
								end
							end)
						end
					end
				end
			end
		end)
	end

	function Culler.stop()
		if EPIX.State.Culler_Conn then
			EPIX.State.Culler_Conn:Disconnect()
			EPIX.State.Culler_Conn = nil
		end
	end

	EPIX.Culler = Culler

		if not savedLighting then
			savedLighting = {
				Ambient = Lighting.Ambient,
				OutdoorAmbient = Lighting.OutdoorAmbient,
				ClockTime = Lighting.ClockTime,
				FogEnd = Lighting.FogEnd,
				Brightness = Lighting.Brightness,
			}
		end
		pcall(function()
			if CONFIG.Visuals.Fullbright then
				Lighting.Ambient = CONFIG.Visuals.Ambient
				Lighting.OutdoorAmbient = CONFIG.Visuals.OutdoorAmb
				Lighting.Brightness = 3
			end
			if CONFIG.Visuals.NoFog then
				Lighting.FogEnd = 1e6
				Lighting.FogStart = 1e6
			end
			Lighting.ClockTime = CONFIG.Visuals.ClockTime
		end)
	end

	function Visuals.restore()
		if not savedLighting then return end
		pcall(function()
			Lighting.Ambient = savedLighting.Ambient
			Lighting.OutdoorAmbient = savedLighting.OutdoorAmbient
			Lighting.ClockTime = savedLighting.ClockTime
			Lighting.FogEnd = savedLighting.FogEnd
			Lighting.Brightness = savedLighting.Brightness
		end)
		savedLighting = nil
	end

	EPIX.Visuals = Visuals

		local ok, v = pcall(function()

		end)
		return ok and v or 0
	end

		EPIX.State.Optimize_Conn = RunService.Heartbeat:Connect(function()
			if CONFIG.Optimize.LowMem and collectgarbage then
				collectgarbage("collect")
			end
		end)
	end

	function Optimize.stop()
		if EPIX.State.Optimize_Conn then
			EPIX.State.Optimize_Conn:Disconnect()
			EPIX.State.Optimize_Conn = nil
		end
	end

	function Optimize.fps()
		return math.floor(1 / (RunService.RenderStepped:Wait() or 0.016))
	end

			Size = UDim2.new(1, 0, 0, 88), BackgroundColor3 = T.Card,
			BorderSizePixel = 0, ZIndex = 11,
		}, Pages[CurrentTab])
		corner(c, 10)
		stroke(c, Color3.fromRGB(90, 70, 140), 1, 0.72)

		new("TextLabel", {
			Size = UDim2.new(1, 0, 0, 18), Position = UDim2.fromOffset(14, 10),
			BackgroundTransparency = 1, Text = label, TextColor3 = T.Text,
			Font = Enum.Font.GothamBold, TextSize = 13,
			TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 12,
		}, c)

		local row = new("Frame", {
			Size = UDim2.new(1, -28, 0, 34), Position = UDim2.fromOffset(14, 40),
			BackgroundTransparency = 1, ZIndex = 12,
		}, c)
		new("UIListLayout", {
			FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 6),
		}, row)

		local buttons = {}
		local function refresh()
			for k, b in pairs(buttons) do
				b.BackgroundColor3 = (k == CONFIG.Selected and CONFIG.Selected) and T.Accent or T.Card
				b.TextColor3 = (k == CONFIG.Selected and CONFIG.Selected) and T.Text or T.TextDim
			end
		end

		CONFIG.Selected = CONFIG.Selected or {}
		CONFIG.Selected[label] = default

		for _, opt in ipairs(options) do
			local b = new("TextButton", {
				Size = UDim2.fromOffset(0, 30), AutomaticSize = Enum.AutomaticSize.X,
				BackgroundColor3 = T.Card, Text = opt,
				TextColor3 = T.TextDim, Font = Enum.Font.Gotham,
				TextSize = 11, BorderSizePixel = 0, AutoButtonColor = false, ZIndex = 13,
			}, row)
			corner(b, 8)
			pad(b, 0, 12, 0, 12)
			buttons[opt] = b
			b.MouseButton1Click:Connect(function()
				CONFIG.Selected[label] = opt
				refresh()
				pcall(cb, opt)
			end)
		end
		refresh()
		return c
	end

	EPIX.Presets = Presets

	--========================================================================
	-- build
	--========================================================================
	GUI.Screen = Screen
	GUI.Main = Main
	GUI.Content = Content
	GUI.TabBar = TabBar
	GUI.Header = Header
	GUI.Logo = Logo
	GUI.Footer = Footer

	GUI.registerPage = registerPage
	GUI.makeTab = makeTab
	GUI.selectTab = selectTab
	GUI.Pages = Pages
	GUI.TabBtns = TabBtns

	GUI.Backdrop = Backdrop
	GUI.BtnMin = BtnMin
	GUI.BtnExp = BtnExp
	GUI.BtnClose = BtnClose

	EPIX.GUI = GUI
	EPIX.State.GuiRefs.Toggle = function()
		Main.Visible = not Main.Visible
	end

	print("[EPIX] GUI builder ready")
end)

	--========================================================================
	-- WINDOW CONTROLS
	--========================================================================
	local Main = GUI.Main
	local drag = { active = false, startPos = nil, startFrame = nil }

	local function beginDrag(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
		or i.UserInputType == Enum.UserInputType.Touch then
			drag.active = true
			drag.startPos = i.Position
			drag.startFrame = Main.Position
		end
	end
	GUI.Header.InputBegan:Connect(beginDrag)
	GUI.Logo.InputBegan:Connect(beginDrag)
	EPIX.S.UserInputService.InputChanged:Connect(function(i)
		if not drag.active then return end
		if i.UserInputType == Enum.UserInputType.MouseMovement
		or i.UserInputType == Enum.UserInputType.Touch then
			local d = i.Position - drag.startPos
			Main.Position = UDim2.new(
				drag.startFrame.X.Scale, drag.startFrame.X.Offset + d.X,
				drag.startFrame.Y.Scale, drag.startFrame.Y.Offset + d.Y)
		end
	end)
	EPIX.S.UserInputService.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1
		or i.UserInputType == Enum.UserInputType.Touch then
			drag.active = false
		end
	end)
